
test "h precedence" = calc("2+3*4") == Ok(14)
test "h parens" = calc("(2+3)*4") == Ok(20)
test "h spaces" = calc(" 12 / 5 ") == Ok(2)
test "h unary" = calc("-3 + 5") == Ok(2)
test "h unary in mul" = calc("2*-3") == Ok(-6)
test "h unary paren" = calc("-(2+3)*2") == Ok(-10)
test "h trunc toward zero" = calc("-7/2") == Ok(-3)
test "h chain sub" = calc("7-10/3") == Ok(4)
test "h nested" = calc("((2))+((1+1)*(3-1))") == Ok(6)
test "h div zero" = match calc("10/(5-5)") { Err(m) => contains(m, "division by zero"), Ok(_) => false, }
test "h dangling op" = match calc("2+") { Err(m) => contains(m, "syntax"), Ok(_) => false, }
test "h unclosed" = match calc("(1+2") { Err(m) => contains(m, "syntax"), Ok(_) => false, }
test "h juxtapose" = match calc("1 2") { Err(m) => contains(m, "syntax"), Ok(_) => false, }
test "h empty" = match calc("") { Err(m) => contains(m, "syntax"), Ok(_) => false, }
