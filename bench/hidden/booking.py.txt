
def _expect_err(existing, s, e, sub):
    try:
        book(existing, s, e)
        return False
    except ValueError as err:
        return sub in str(err)

assert book([], 1, 3) == [(1, 3)], "first"
assert book([(1, 3)], 3, 5) == [(1, 5)], "touch right merges"
assert book([(1, 3), (5, 7)], 3, 5) == [(1, 7)], "chain merge"
assert book([(4, 6)], 1, 4) == [(1, 6)], "merge left"
assert book([(0, 10)], 10, 12) == [(0, 12)], "extend at end"
assert book([(5, 7)], 1, 3) == [(1, 3), (5, 7)], "sorted insert"
assert _expect_err([(1, 3)], 2, 4, "overlap"), "overlap"
assert _expect_err([(0, 10)], 3, 4, "overlap"), "contained"
assert _expect_err([(4, 6)], 4, 4, "invalid"), "empty range"
assert _expect_err([], 5, 2, "invalid"), "reversed"
