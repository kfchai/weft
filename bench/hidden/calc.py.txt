
def _expect_err(s, sub):
    try:
        calc(s)
        return False
    except ValueError as e:
        return sub in str(e)

assert calc("2+3*4") == 14, "precedence"
assert calc("(2+3)*4") == 20, "parens"
assert calc(" 12 / 5 ") == 2, "spaces"
assert calc("-3 + 5") == 2, "unary"
assert calc("2*-3") == -6, "unary in mul"
assert calc("-(2+3)*2") == -10, "unary paren"
assert calc("-7/2") == -3, "trunc toward zero"
assert calc("7-10/3") == 4, "chain sub"
assert calc("((2))+((1+1)*(3-1))") == 6, "nested"
assert _expect_err("10/(5-5)", "division by zero"), "div zero"
assert _expect_err("2+", "syntax"), "dangling op"
assert _expect_err("(1+2", "syntax"), "unclosed"
assert _expect_err("1 2", "syntax"), "juxtapose"
assert _expect_err("", "syntax"), "empty"
print("ALL HIDDEN TESTS PASS")
